const https = require('https'); function sendJson(status, payload) { return { statusCode: status, headers: { 'Content-Type': 'application/json; charset=utf-8', 'Access-Control-Allow-Origin': '*', 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type' }, body: JSON.stringify(payload) }; } function encodeGitHubPath(targetPath) { return targetPath.split('/').map(encodeURIComponent).join('/'); } function githubApiRequest(options, body) { return new Promise((resolve, reject) => { const req = https.request(options, res => { let data = ''; res.on('data', chunk => { data += chunk; }); res.on('end', () => { resolve({ status: res.statusCode || 500, headers: res.headers, body: data }); }); }); req.on('error', reject); if (body) { req.write(body); } req.end(); }); } function parseJsonSafe(body) { try { return body ? JSON.parse(body) : {}; } catch (err) { return null; } } async function getGitHubFileSha(owner, repo, branch, path, token) { const encodedPath = encodeGitHubPath(path); const options = { hostname: 'api.github.com', path: `/repos/${encodeURIComponent(owner)}/${encodeURIComponent(repo)}/contents/${encodedPath}?ref=${encodeURIComponent(branch)}`, method: 'GET', headers: { 'User-Agent': 'netlify-function', 'Accept': 'application/vnd.github+json', 'Authorization': `Bearer ${token}` } }; const response = await githubApiRequest(options); if (response.status === 200) { const json = parseJsonSafe(response.body); return (json && json.sha) ? json.sha : null; } return null; } exports.handler = async function(event) { if (event.httpMethod === 'OPTIONS') { return sendJson(204, {}); } if (event.httpMethod !== 'POST') { return sendJson(405, { error: 'Method Not Allowed' }); } const GITHUB_TOKEN = process.env.GITHUB_TOKEN; const GITHUB_OWNER = process.env.GITHUB_OWNER; const GITHUB_REPO = process.env.GITHUB_REPO; const GITHUB_BRANCH = process.env.GITHUB_BRANCH || 'main'; const GITHUB_PAGES_BASE_URL = process.env.GITHUB_PAGES_BASE_URL || ''; if (!GITHUB_TOKEN || !GITHUB_OWNER || !GITHUB_REPO) { return sendJson(500, { error: 'Missing GitHub configuration: set GITHUB_TOKEN, GITHUB_OWNER and GITHUB_REPO in Netlify environment.' }); } let payload; try { payload = event.body ? JSON.parse(event.body) : {}; } catch (err) { return sendJson(400, { error: 'Invalid JSON payload' }); } const targetPath = String(payload.path || payload.fileName || '').trim(); const content = String(payload.content || '').trim(); const commitMessage = String(payload.commitMessage || `Upload vidéo ${targetPath}`).trim(); if (!targetPath) { return sendJson(400, { error: 'Missing file path' }); } if (!content) { return sendJson(400, { error: 'Missing base64 file content' }); } if (targetPath.includes('..')) { return sendJson(400, { error: 'Invalid file path' }); } const normalizedPath = targetPath.replace(/^\/+/, '').replace(/\\/g, '/'); const sha = await getGitHubFileSha(GITHUB_OWNER, GITHUB_REPO, GITHUB_BRANCH, normalizedPath, GITHUB_TOKEN); const body = { message: commitMessage, content, branch: GITHUB_BRANCH }; if (sha) body.sha = sha; const options = { hostname: 'api.github.com', path: `/repos/${encodeURIComponent(GITHUB_OWNER)}/${encodeURIComponent(GITHUB_REPO)}/contents/${encodeGitHubPath(normalizedPath)}`, method: 'PUT', headers: { 'User-Agent': 'netlify-function', 'Accept': 'application/vnd.github+json', 'Authorization': `Bearer ${GITHUB_TOKEN}`, 'Content-Type': 'application/json', 'Content-Length': Buffer.byteLength(JSON.stringify(body)) } }; try { const response = await githubApiRequest(options, JSON.stringify(body)); const result = parseJsonSafe(response.body); if (!result && response.body) { return sendJson(response.status >= 400 ? response.status : 502, { error: 'GitHub returned invalid JSON', details: response.body }); } if (response.status >= 400) { return sendJson(response.status, { error: result.message || 'GitHub upload failed', details: result }); } const downloadUrl = result.content && result.content.download_url ? result.content.download_url : null; const pagesUrl = GITHUB_PAGES_BASE_URL ? `${GITHUB_PAGES_BASE_URL.replace(/\/$/, '')}/${normalizedPath}` : `https://${GITHUB_OWNER.toLowerCase() === `${GITHUB_OWNER}.github.io` ? GITHUB_OWNER : `${GITHUB_OWNER}.github.io/${GITHUB_REPO}`}/${normalizedPath}`; return sendJson(200, { fileUrl: downloadUrl || pagesUrl, pagesUrl, rawUrl: downloadUrl }); } catch (err) { return sendJson(502, { error: err.message }); } };
